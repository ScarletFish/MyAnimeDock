# Tauri + SQLite/Prisma 迁移计划

基于代码库分析和你的技术决策，以下是详细迁移计划：

---

## 🎯 核心决策确认

| 决策项 | 选择 |
|--------|------|
| Tauri 策略 | **Sidecar 方案** — Node.js server.js 作为 sidecar 进程，前端保持 HTTP API 调用 |
| 数据库路径 | **默认 exe 同级目录 + 设置页可自定义** |
| Prisma Schema | **library/episodes/memories/playSessions 规范化建表；scannedTree/config 保持 JSON 字段** |
| 数据迁移 | **一次性迁移脚本**（启动时检测旧 JSON 自动迁移） |
| 前端通信 | **保持 HTTP API**（fetch() 完全不变） |

## 🗄️ 阶段 1：Prisma Schema 设计（预计 3-4 小时）

### 1.1 核心模型设计

```prisma
// prisma/schema.prisma
generator client {
  provider = "prisma-client-js"
}

datasource db {
  provider = "sqlite"
  url      = env("DATABASE_URL")
}

// ===== 规范化表 =====

model Anime {
  id            String    @id @default(cuid())
  folderPath    String    @unique
  folderName    String
  title         String
  season        Int?
  importedAt    DateTime  @default(now())
  downloaded    Boolean   @default(true)
  bangumiId     Int?      @unique
  bangumiTitle  String?
  bangumiTitleJp String?
  summary       String?   @db.Text
  coverUrl      String?
  localCover    String?
  rating        Float?
  source        String?   // "bangumi" | "tmdb"
  pinyinTitle   String?
  metadata      Json?     // 扩展字段（兼容旧数据）
  
  episodes      Episode[]
  playSessions  PlaySession[]
  memories      Memory[]
  
  @@index([bangumiId])
  @@index([title])
  @@index([pinyinTitle])
}

model Episode {
  id          String   @id @default(cuid())
  animeId     String
  number      Int
  filePath    String   @unique
  fileName    String
  fileSize    BigInt
  duration    Int?     // 秒
  watched     Boolean  @default(false)
  progress    Float    @default(0)  // 0-1
  watchedAt   DateTime?
  
  anime       Anime    @relation(fields: [animeId], references: [id], onDelete: Cascade)
  
  @@unique([animeId, number])
  @@index([animeId])
  @@index([filePath])
}

model PlaySession {
  id            String   @id @default(cuid())
  animeId       String
  episodeNumber Int
  sessionId     String   @unique
  startTime     DateTime
  endTime       DateTime?
  duration      Int      @default(0)   // 内容进度秒数
  clockTime     Int      @default(0)   // 挂钟秒数
  progressStart Int      @default(0)
  
  anime         Anime    @relation(fields: [animeId], references: [id], onDelete: Cascade)
  
  @@index([animeId])
  @@index([sessionId])
  @@index([startTime])
}

model Memory {
  id           String   @id @default(cuid())
  animeId      String   @unique
  title        String
  bangumiId    Int?
  bangumiTitle String?
  rating       Float?
  thoughts     String?  @db.Text
  notes        String?  @db.Text
  watchedAt    DateTime @default(now())
  coverLocal   String?
  
  anime        Anime    @relation(fields: [animeId], references: [id], onDelete: Cascade)
  
  @@index([animeId])
  @@index([watchedAt])
}

// ===== JSON 存储表（保持灵活性）=====

model ScannedTree {
  id        String   @id @default(cuid())
  data      Json     // 扁平 leaf 数组，含 parentChain
  updatedAt DateTime @updatedAt
}

model Config {
  id        String   @id @default("singleton")
  data      Json     // 完整配置对象
  updatedAt DateTime @updatedAt
}

// ===== 迁移追踪 =====

model MigrationLog {
  id          String   @id @default(cuid())
  version     String   @unique
  appliedAt   DateTime @default(now())
  description String?
}
```

### 1.2 关键设计说明

| 表 | 设计理由 |
|----|----------|
| `Anime` | 核心实体，folderPath 唯一索引防重复导入 |
| `Episode` | 规范化存储，支持单集进度/时长/观看状态 |
| `PlaySession` | 完整保留 mpv 追踪数据，支持 90 天统计图表 |
| `Memory` | 归档页数据，animeId 唯一（一番一档） |
| `ScannedTree` | 单行 JSON，扫描结果不频繁查询单条 |
| `Config` | 单例模式，整个配置对象序列化存储 |

## 🔄 阶段 2：数据迁移脚本（预计 2-3 小时）

### 2.1 迁移脚本 `scripts/migrate-to-sqlite.js`

```javascript
// scripts/migrate-to-sqlite.js
const fs = require('fs');
const path = require('path');
const { PrismaClient } = require('@prisma/client');

const prisma = new PrismaClient();

const DATA_PATH = path.join(__dirname, '..', 'anime-data.json');
const BACKUP_PATH = path.join(__dirname, '..', 'anime-data.json.pre-sqlite-backup');

async function migrate() {
  console.log('[Migration] Starting anime-data.json → SQLite migration...');
  
  // 1. 读取旧数据
  if (!fs.existsSync(DATA_PATH)) {
    console.log('[Migration] No anime-data.json found, skipping.');
    return;
  }
  
  const raw = fs.readFileSync(DATA_PATH, 'utf-8');
  const oldData = JSON.parse(raw);
  
  // 2. 备份原文件
  fs.copyFileSync(DATA_PATH, BACKUP_PATH);
  console.log(`[Migration] Backed up to ${BACKUP_PATH}`);
  
  // 3. 事务迁移
  await prisma.$transaction(async (tx) => {
    // 3.1 迁移 Anime + Episodes
    for (const anime of oldData.library || []) {
      const created = await tx.anime.create({
        data: {
          id: anime.id,  // 保持原 ID 兼容
          folderPath: anime.folderPath,
          folderName: anime.folderName,
          title: anime.title,
          season: anime.season,
          importedAt: new Date(anime.importedAt),
          downloaded: anime.downloaded,
          bangumiId: anime.bangumiId,
          bangumiTitle: anime.bangumiTitle,
          bangumiTitleJp: anime.bangumiTitleJp,
          summary: anime.summary,
          coverUrl: anime.coverUrl,
          localCover: anime.localCover,
          rating: anime.rating,
          source: anime.source,
          pinyinTitle: anime.pinyinTitle,
          metadata: {
            // 保留不在 schema 中的字段
            episodes: anime.episodes?.map(e => ({
              number: e.number,
              filePath: e.filePath,
              fileName: e.fileName,
              fileSize: e.fileSize,
              duration: e.duration,
              watched: e.watched,
              progress: e.progress,
            })) || [],
          },
        },
      });
      
      // 3.2 批量创建 Episodes
      if (anime.episodes?.length) {
        await tx.episode.createMany({
          data: anime.episodes.map(e => ({
            animeId: created.id,
            number: e.number,
            filePath: e.filePath,
            fileName: e.fileName,
            fileSize: BigInt(e.fileSize),
            duration: e.duration,
            watched: e.watched,
            progress: e.progress,
            watchedAt: e.watched ? new Date() : null,
          })),
        });
      }
    }
    console.log(`[Migration] Migrated ${oldData.library?.length || 0} anime`);
    
    // 3.3 迁移 Memories
    for (const mem of oldData.memories || []) {
      await tx.memory.upsert({
        where: { animeId: mem.animeId },
        create: {
          animeId: mem.animeId,
          title: mem.title,
          bangumiId: mem.bangumiId,
          bangumiTitle: mem.bangumiTitle,
          rating: mem.rating,
          thoughts: mem.thoughts,
          notes: mem.notes,
          watchedAt: new Date(mem.watchedAt),
          coverLocal: mem.coverLocal,
        },
        update: {
          rating: mem.rating,
          thoughts: mem.thoughts,
          notes: mem.notes,
          coverLocal: mem.coverLocal,
        },
      });
    }
    console.log(`[Migration] Migrated ${oldData.memories?.length || 0} memories`);
    
    // 3.4 迁移 PlaySessions
    if (oldData.playSessions?.length) {
      await tx.playSession.createMany({
        data: oldData.playSessions.map(s => ({
          animeId: s.animeId,
          episodeNumber: s.episodeNumber,
          sessionId: s.sessionId,
          startTime: new Date(s.startTime),
          endTime: s.endTime ? new Date(s.endTime) : null,
          duration: s.duration || 0,
          clockTime: s.clockTime || 0,
          progressStart: s.progressStart || 0,
        })),
        skipDuplicates: true,
      });
      console.log(`[Migration] Migrated ${oldData.playSessions.length} play sessions`);
    }
    
    // 3.5 迁移 ScannedTree
    if (oldData.scannedTree?.length) {
      await tx.scannedTree.upsert({
        where: { id: 'current' },
        create: { id: 'current', data: oldData.scannedTree },
        update: { data: oldData.scannedTree },
      });
      console.log(`[Migration] Migrated scannedTree (${oldData.scannedTree.length} nodes)`);
    }
    
    // 3.6 迁移 Config
    const configPath = path.join(__dirname, '..', 'config.json');
    if (fs.existsSync(configPath)) {
      const configRaw = fs.readFileSync(configPath, 'utf-8');
      const config = JSON.parse(configRaw);
      await tx.config.upsert({
        where: { id: 'singleton' },
        create: { id: 'singleton', data: config },
        update: { data: config },
      });
      console.log('[Migration] Migrated config');
    }
    
    // 3.7 记录迁移版本
    await tx.migrationLog.create({
      data: {
        version: '1.0.0',
        description: 'Initial migration from anime-data.json to SQLite',
      },
    });
  });
  
  console.log('[Migration] ✅ Migration completed successfully!');
  console.log('[Migration] You can now delete anime-data.json (backed up)');
}

migrate()
  .catch(e => {
    console.error('[Migration] ❌ Failed:', e);
    process.exit(1);
  })
  .finally(() => prisma.$disconnect());
```

### 2.2 集成到启动流程

```javascript
// server.js 顶部新增
const { PrismaClient } = require('@prisma/client');
const fs = require('fs');
const path = require('path');

const prisma = new PrismaClient();
const DATA_PATH = path.join(APP_DIR, 'anime-data.json');
const MIGRATION_MARKER = path.join(APP_DIR, '.migrated-to-sqlite');

async function ensureMigrated() {
  if (fs.existsSync(MIGRATION_MARKER)) return;
  if (!fs.existsSync(DATA_PATH)) {
    // 首次运行，创建标记文件
    fs.writeFileSync(MIGRATION_MARKER, Date.now().toString());
    return;
  }
  
  console.log('[Startup] Detected legacy anime-data.json, running migration...');
  const { migrate } = require('./scripts/migrate-to-sqlite');
  await migrate();
  fs.writeFileSync(MIGRATION_MARKER, Date.now().toString());
}

// 在 server.listen() 前调用
await ensureMigrated();
```

### 3.2 `src-tauri/src/main.rs` — Sidecar 启动

```rust
// src-tauri/src/main.rs
use tauri::Manager;
use std::process::Command;
use std::path::PathBuf;

fn main() {
    tauri::Builder::default()
        .setup(|app| {
            let handle = app.handle();
            
            // 启动 Node.js sidecar
            let sidecar_path = get_sidecar_path(&handle)?;
            println!("Starting sidecar: {:?}", sidecar_path);
            
            let mut child = Command::new("node")
                .arg(&sidecar_path)
                .current_dir(sidecar_path.parent().unwrap())
                .spawn()
                .expect("Failed to start Node.js sidecar");
            
            // 存储 child 以便退出时清理
            handle.manage(SidecarProcess(child));
            
            // 等待 server 就绪（轮询 /api/config）
            std::thread::spawn(move || {
                wait_for_server_ready();
            });
            
            Ok(())
        })
        .on_window_event(|window, event| {
            if let tauri::WindowEvent::CloseRequested { .. } = event {
                // 优雅关闭 sidecar
                if let Some(sidecar) = window.try_state::<SidecarProcess>() {
                    sidecar.0.kill().ok();
                }
            }
        })
        .run(tauri::generate_context!())
        .expect("error while running tauri application");
}

fn get_sidecar_path(handle: &tauri::AppHandle) -> Result<PathBuf, Box<dyn std::error::Error>> {
    let resource_dir = handle.path().resource_dir()?;
    Ok(resource_dir.join("server").join("server.js"))
}

fn wait_for_server_ready() {
    use std::time::Duration;
    for _ in 0..30 {
        if let Ok(resp) = ureq::get("http://localhost:3456/api/config").call() {
            if resp.status() == 200 {
                println!("Sidecar server ready");
                return;
            }
        }
        std::thread::sleep(Duration::from_millis(500));
    }
    eprintln!("Warning: Sidecar server did not become ready in time");
}

struct SidecarProcess(std::process::Child);
```

## 🔧 阶段 4：Server.js 数据层重构（预计 4-5 小时）

### 4.1 新建 `server/db.js` — Prisma 封装

```javascript
// server/db.js
const { PrismaClient } = require('@prisma/client');
const path = require('path');

let prisma = null;

function getPrisma() {
  if (!prisma) {
    const dbPath = process.env.DATABASE_URL || 
      `file:${path.join(process.cwd(), 'anime.db')}`;
    prisma = new PrismaClient({
      datasources: { db: { url: dbPath } },
      log: process.env.NODE_ENV === 'development' ? ['query', 'error', 'warn'] : ['error'],
    });
  }
  return prisma;
}

// ===== Anime CRUD =====
async function getAnimeList() {
  return getPrisma().anime.findMany({
    include: { episodes: { orderBy: { number: 'asc' } } },
    orderBy: { importedAt: 'desc' },
  });
}

async function getAnimeById(id) {
  return getPrisma().anime.findUnique({
    where: { id },
    include: { episodes: { orderBy: { number: 'asc' } } },
  });
}

async function createAnime(data) {
  const { episodes, ...animeData } = data;
  return getPrisma().anime.create({
    data: {
      ...animeData,
      episodes: episodes ? { create: episodes } : undefined,
    },
    include: { episodes: true },
  });
}

async function updateAnime(id, data) {
  return getPrisma().anime.update({
    where: { id },
    data,
    include: { episodes: true },
  });
}

async function deleteAnime(id) {
  return getPrisma().anime.delete({ where: { id } });
}

// ===== Episodes =====
async function updateEpisode(animeId, episodeNumber, data) {
  return getPrisma().episode.update({
    where: { animeId_number: { animeId, number: episodeNumber } },
    data,
  });
}

// ===== PlaySessions =====
async function createPlaySession(data) {
  return getPrisma().playSession.create({ data });
}

async function updatePlaySession(sessionId, data) {
  return getPrisma().playSession.update({
    where: { sessionId },
    data,
  });
}

async function getPlaySessions(animeId) {
  return getPrisma().playSession.findMany({
    where: { animeId, endTime: { not: null } },
    orderBy: { startTime: 'asc' },
  });
}

// ===== Memories =====
async function getMemories() {
  return getPrisma().memory.findMany({
    orderBy: { watchedAt: 'desc' },
  });
}

async function upsertMemory(data) {
  return getPrisma().memory.upsert({
    where: { animeId: data.animeId },
    create: data,
    update: data,
  });
}

// ===== ScannedTree =====
async function getScannedTree() {
  const record = await getPrisma().scannedTree.findUnique({ where: { id: 'current' } });
  return record?.data || [];
}

async function saveScannedTree(tree) {
  return getPrisma().scannedTree.upsert({
    where: { id: 'current' },
    create: { id: 'current', data: tree },
    update: { data: tree },
  });
}

// ===== Config =====
async function getConfig() {
  const record = await getPrisma().config.findUnique({ where: { id: 'singleton' } });
  return record?.data || {};
}

async function saveConfig(cfg) {
  return getPrisma().config.upsert({
    where: { id: 'singleton' },
    create: { id: 'singleton', data: cfg },
    update: { data: cfg },
  });
}

module.exports = {
  getPrisma,
  getAnimeList,
  getAnimeById,
  createAnime,
  updateAnime,
  deleteAnime,
  updateEpisode,
  createPlaySession,
  updatePlaySession,
  getPlaySessions,
  getMemories,
  upsertMemory,
  getScannedTree,
  saveScannedTree,
  getConfig,
  saveConfig,
};
```

## ⚙️ 阶段 5：配置系统统一（预计 1-2 小时）

### 5.1 统一配置管理

```javascript
// server/config.js
const db = require('./db');
const path = require('path');

const DEFAULT_CONFIG = { 
  mediaDir: '', 
  playerMode: 'system', 
  mpvPath: 'mpv', 
  theme: 'dark',
  autoMarkWatched: true,
  apiSources: [{ type: 'bangumi', url: 'https://api.bangumi.one', key: '' }],
  databasePath: '',  // 用户自定义数据库路径
};

let configCache = null;

async function loadConfig() {
  if (configCache) return configCache;
  
  const stored = await db.getConfig();
  configCache = { ...DEFAULT_CONFIG, ...stored };
  
  // 处理 databasePath：支持相对路径（相对于 exe 目录）
  if (configCache.databasePath && !path.isAbsolute(configCache.databasePath)) {
    configCache.databasePath = path.join(process.cwd(), configCache.databasePath);
  }
  
  return configCache;
}

async function saveConfig(cfg) {
  configCache = { ...configCache, ...cfg };
  await db.saveConfig(configCache);
  // 如果 databasePath 变更，需重新初始化 Prisma
  if (cfg.databasePath) {
    process.env.DATABASE_URL = `file:${path.isAbsolute(cfg.databasePath) ? cfg.databasePath : path.join(process.cwd(), cfg.databasePath)}`;
    // 触发 Prisma 重连（下次 getPrisma() 会用新 URL）
  }
  return configCache;
}

module.exports = { loadConfig, saveConfig, DEFAULT_CONFIG };
```
